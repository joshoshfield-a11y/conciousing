\begin{abstract}
We formalize \emph{conciousing} --- the continuous, enacted process of bringing forth a world for a subject --- as a stochastic dynamical system $C_t = (b_t, \pi_t, \Gvec_t, \Pvec_t, \iota_t)$ over a hierarchy of timescales $\tauvec$. We derive differentiable estimators for each component, prove integration bounds linking variational free energy, mutual information, and $\Phi$-like measures, and provide a reference implementation for real-time conciousing profiling in hierarchical active-inference agents.
\end{abstract}

\section{Foundations: The Conciousing State}
\label{sec:foundations}

\begin{definition}[Agent Tuple]
An \emph{agent} is a tuple
\[
\mathcal{A} = (\Sset, \Aset, \Oset, T, R, \pi, \Mcal, \tauvec)
\]
where
\begin{itemize}
    \item $\Sset$ latent states, $\Aset$ actions, $\Oset$ observations
    \item $T(s' \given s,a)$ transition density, $R(s,a)$ reward / intrinsic value
    \item $\pi(a \given s)$ policy, $\Mcal$ generative world-model $p_\theta(o_t, s_t \given o_{<t}, a_{<t})$
    \item $\tauvec = (\tau_1, \dots, \tau_K)$ with $\tau_1 \ll \tau_2 \ll \dots \ll \tau_K$ a hierarchy of temporal horizons
\end{itemize}
\end{definition}

\begin{definition}[Conciousing State $C_t$]
The \emph{conciousing state} at time $t$ is the 5-tuple
\begin{equation}
C_t = \bigl( b_t,\; \pi_t,\; \Gvec_t,\; \Pvec_t,\; \iota_t \bigr)
\label{eq:Ct}
\end{equation}
where
\begin{align}
b_t(s) &:= p(s_t = s \given o_{\le t}, a_{<t}) && \text{(belief state / posterior)} \label{eq:belief} \\
\pi_t(a) &:= \pi(a \given b_t) && \text{(policy conditioned on belief)} \label{eq:policy} \\
\Gvec_t &:= \{ g^{(k)}_t \}_{k=1}^K, \quad g^{(k)}_t \in \Delta(\Sset) && \text{(goal / target distributions per timescale)} \label{eq:goals} \\
\Pvec_t &:= \{ p^{(k)}_t \}_{k=1}^K, \quad p^{(k)}_t(s_{t:t+\tau_k} \given s_t, \pi) && \text{(counterfactual simulations)} \label{eq:sims} \\
\iota_t &:= \Iset(b_t; \pi_t, \Gvec_t, \Pvec_t) && \text{(integration measure)} \label{eq:iota}
\end{align}
\end{definition}

\begin{definition}[Conciousing Trajectory]
The \emph{conciousing trajectory} over interval $[t_0, t_1]$ is the continuous-time limit (or discrete sequence) $\Ccal_{t_0:t_1} = \{ C_t \}_{t=t_0}^{t_1}$.
The \emph{conciousing intensity} over a window $\Delta$ is
\[
\Phi_\Delta(C) := \frac{1}{\Delta} \sum_{t=t_0}^{t_0+\Delta} \iota_t \,.
\]
\end{definition}

\section{Hierarchical Active Inference as Conciousing Dynamics}
\label{sec:hai}

We instantiate the agent as a \emph{Hierarchical Active Inference} (HAI) system \cite{parr2022active, friston2017active}. At each level $k=1\dots K$:
\begin{align}
\text{Generative model:} \quad & p^{(k)}(o^{(k)}, s^{(k)} \given \pi^{(k)}) \\
\text{Variational posterior:} \quad & q^{(k)}(s^{(k)} \given o^{(\le k)}) =: b^{(k)}_t \\
\text{Expected Free Energy:} \quad & G^{(k)}(\pi^{(k)}) = \E_{q^{(k)}}[ -\ln p^{(k)}(o^{(k)} \given s^{(k)}) ] + \KL[ q^{(k)}(s^{(k)}) \parallel p^{(k)}(s^{(k)}) ] \\
\text{Policy (intention):} \quad & \pi^{(k)}(\pi^{(k)} \given b^{(k)}) \propto \exp( - G^{(k)}(\pi^{(k)}) ) \\
\text{Goal prior:} \quad & g^{(k)}_t = p^{(k)}(o^{(k)} \given \text{preferences}) \quad \text{(often a target density over observations)}
\end{align}
The \emph{conciousing state at level $k$} is $C^{(k)}_t = (b^{(k)}_t, \pi^{(k)}_t, g^{(k)}_t, \Pvec^{(k)}_t, \iota^{(k)}_t)$.
The \emph{global conciousing state} aggregates across levels:
\[
C_t = \mathrm{Agg}\bigl( \{ C^{(k)}_t \}_{k=1}^K \bigr), \quad \text{e.g., } \iota_t = \sum_k w_k \iota^{(k)}_t + \MI( \{b^{(k)}\}; \{\pi^{(k)}\} ).
\]

\section{Differentiable Estimators for Real-Time Profiling}
\label{sec:estimators}

All estimators below are designed to be computed \emph{online} with $\mathcal{O}(1)$ amortized cost per agent step, using reparameterization gradients where needed.

\subsection{Belief Entropy $\Ent[b_t]$}
For a Gaussian belief $b_t = \mathcal{N}(\mu_t, \Sigma_t)$:
\[
\Ent[b_t] = \frac{1}{2} \ln \det(2\pi e \Sigma_t).
\]
For categorical $b_t \in \Delta^{|\Sset|-1}$: $\Ent[b_t] = -\sum_s b_t(s) \ln b_t(s)$.
For particle beliefs $\{s^{(i)}_t, w^{(i)}_t\}_{i=1}^N$: use \emph{Kozachenko--Leonenko} $k$-NN estimator \cite{kozachenko1987sample}.

\subsection{Policy Entropy $\Ent[\pi_t]$}
\[
\Ent[\pi_t] = -\sum_a \pi_t(a) \ln \pi_t(a) \quad \text{or} \quad \frac{1}{2}\ln\det(2\pi e \Sigma_{\pi_t}) \text{ for Gaussian policies}.
\]

\subsection{Goal Stack Depth $\|\Gvec_t\|_0$}
Simply $K_t = |\{ k : \KL(g^{(k)}_t \parallel \mathrm{unif}) > \epsilon \}|$.

\subsection{Simulation Horizon $\|\Pvec_t\|_1$}
Total imagined steps across all levels: $H_t = \sum_{k=1}^K \tau_k \cdot \mathbb{I}[\text{level $k$ active}]$.

\subsection{Integration Proxy $\iota_t$}
We provide three estimators of increasing fidelity:

\begin{algorithm}[H]
\caption{Differentiable Integration Estimators}
\label{alg:iota}
\textbf{Input:} $b_t$ (belief), $\pi_t$ (policy), $\{g^{(k)}_t\}$, $\{p^{(k)}_t\}$ \\
\textbf{Output:} $\iota_t \in \R_{\ge 0}$
\begin{enumerate}
    \item \textbf{MI-Lower-Bound (MINE-style) \cite{belghazi2018mine}:} \\
    Train a critic $T_\psi(b, \pi, \Gvec, \Pvec)$ to maximize
    \[
    \iota_t^{\mathrm{MI}} = \E_{p(b,\pi,\Gvec,\Pvec)}[T_\psi] - \ln \E_{p(b)p(\pi,\Gvec,\Pvec)}[e^{T_\psi}].
    \]
    \item \textbf{Free-Energy Gap (HAI-native):} \\
    \[
    \iota_t^{\mathrm{FE}} = \sum_{k=1}^K \bigl( F^{(k)}_{\mathrm{prior}} - F^{(k)}_{\mathrm{post}} \bigr), \quad
    F^{(k)} = \E_{q^{(k)}}[-\ln p^{(k)}] + \KL[q^{(k)} \parallel p^{(k)}].
    \]
    This is \emph{exactly} the variational free energy minimized by the agent.
    \item \textbf{$\Phi$-Proxy (IIT-style, differentiable \cite{oizumi2014phenomenology}):} \\
    Partition the latent state $s = (s_A, s_B)$. Compute
    \[
    \iota_t^\Phi = \MI( s_A^{(t)}; s_B^{(t+1)} \given \pi_t ) - \MI( s_A^{(t)}; s_B^{(t+1)} \given \pi_t, \text{cut} ).
    \]
    Approximate with Gaussian $\MI = \frac12 \ln \frac{\det\Sigma_{AB}}{\det\Sigma_A \det\Sigma_B}$.
\end{enumerate}
\end{algorithm}

\subsection{Surprise $\mathcal{S}_t$}
\[
\mathcal{S}_t = -\ln p_\theta(o_t \given b_{t-1}) \quad \text{(negative log marginal likelihood)}.
\]
For Gaussian observation model $o_t \sim \mathcal{N}(C s_t, \Sigma_o)$:
\[
\mathcal{S}_t = \frac12 \bigl( (o_t - C\mu_{t\given t-1})\trans \Sigma_o^{-1} (o_t - C\mu_{t\given t-1}) + \ln\det\Sigma_o + d\ln 2\pi \bigr).
\]

\subsection{Counterfactual Breadth $\Ent[\Pvec_t]$}
\[
\Ent[\Pvec_t] = \sum_{k=1}^K \Ent[ p^{(k)}_t(s_{t:t+\tau_k}) ].
\]
For Gaussian latent rollouts, entropy is analytic; for particle rollouts, use K-L estimator.

\section{Theoretical Guarantees}
\label{sec:theory}

\begin{proposition}[Free Energy Bounds Integration]
\label{prop:FE_bound}
For any HAI agent, $\iota_t^{\mathrm{FE}} \ge 0$ and $\iota_t^{\mathrm{FE}} = 0 \iff$ the posterior equals the prior at all levels (no conciousing).
\end{proposition}
\begin{proof}
Each term $F^{(k)}_{\mathrm{prior}} - F^{(k)}_{\mathrm{post}} = \KL[q^{(k)} \parallel p^{(k)}] \ge 0$ by Gibbs' inequality. Sum of non-negatives is non-negative.
\end{proof}

\begin{proposition}[MI Estimator Consistency]
\label{prop:MI_consistency}
If the critic $T_\psi$ has universal approximation capacity, $\iota_t^{\mathrm{MI}} \to \MI(b_t; \pi_t, \Gvec_t, \Pvec_t)$ as optimization converges.
\end{proposition}
\begin{proof}
Direct application of the Donsker-Varadhan representation of KL divergence \cite{donsker1983asymptotic}.
\end{proof}

\begin{theorem}[Conciousing Intensity Monotonicity]
\label{thm:monotonicity}
Under Assumption \ref{ass:ergodic} (ergodic environment), if the agent's policy improves expected free energy monotonically, then $\E[\Phi_\Delta(C)]$ is non-decreasing in $\Delta$ until convergence.
\end{theorem}
\begin{proof}[Sketch]
Policy improvement $\implies$ tighter posteriors $\implies$ larger $\KL[q\parallel p]$ terms $\implies$ larger $\iota_t^{\mathrm{FE}}$. Ergodicity ensures sufficient exploration to update beliefs.
\end{proof}
